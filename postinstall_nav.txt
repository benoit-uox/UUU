#!/bin/bash
# postinstall_nav.sh — Post-installation ESL : couche applicative NAVREP + correctif
# Usage : ./postinstall_nav.sh [--dry-run] [--force]
#   --dry-run : affiche les commandes sans rien modifier
#   --force   : rejoue la post-installation même si déjà effectuée

set -euo pipefail

# =============================================================================
# PARAMETRES — tout se modifie ici, rien d'autre à toucher dans le script.
# Une variable d'environnement passée en ligne de commande reste prioritaire :
#   KEY=/mnt/usb ./postinstall_nav.sh
# =============================================================================

# --- Chemins source (clé USB) ---
KEY="${KEY:-/mnt/key}"                                  # point de montage de la clé
SRC_APP="${SRC_APP:-$KEY/NAVREP}"                       # couche applicative source
SRC_POST="${SRC_POST:-$KEY/postinstall}"                # répertoire des correctifs
SRC_STARTX="${SRC_STARTX:-$SRC_POST/rep/esl-startx.sh}" # correctif à déployer
SRC_SHADOW="${SRC_SHADOW:-$SRC_POST/pwd/shadow_x}"      # fichier shadow de référence

# --- Chemins destination (machine ESL) ---
DST_DATA="${DST_DATA:-/nav/data}"                       # destination applicative
DST_BIN="${DST_BIN:-/opt/esl/bin}"                      # destination du correctif

# --- Comptes et droits ---
APP_USER="${APP_USER:-nav}"
APP_GROUP="${APP_GROUP:-nav}"
DATA_MODE="${DATA_MODE:-755}"                           # droits sur $DST_DATA (racine seule)
SHADOW_MODE="${SHADOW_MODE:-000}"                       # 000 sur OL8 ; 640 si groupe shadow

# --- Traces ---
LOG_DIR="${LOG_DIR:-/var/log}"
STAMP="${STAMP:-/var/lib/postinstall_nav.done}"
# =============================================================================
# Fin des paramètres — le reste du script n'a normalement pas à être modifié.
# =============================================================================

DRY=0; FORCE=0; SHOWCFG=0
for arg in "$@"; do
    case "$arg" in
        --dry-run)     DRY=1 ;;
        --force)       FORCE=1 ;;
        --show-config) SHOWCFG=1 ;;
        --help|-h)     sed -n '2,10p' "${BASH_SOURCE[0]}"; exit 0 ;;
        *) echo "Option inconnue : $arg" >&2
           echo "Usage : $0 [--dry-run] [--force] [--show-config]" >&2; exit 2 ;;
    esac
done

HOST="${HOST:-$(hostname -s)}"
TS="$(date +%Y%m%d_%H%M%S)"
LOG="${LOG:-$LOG_DIR/postinstall_nav_${HOST}_${TS}.log}"

if [[ $SHOWCFG -eq 1 ]]; then
    echo "--- Parametres effectifs ---"
    for v in KEY SRC_APP SRC_POST SRC_STARTX SRC_SHADOW DST_DATA DST_BIN \
             APP_USER APP_GROUP DATA_MODE SHADOW_MODE LOG STAMP; do
        printf '%-12s: %s\n' "$v" "${!v}"
    done
    exit 0
fi

exec > >(tee -a "$LOG") 2>&1

run()  { echo "+ $*"; [[ $DRY -eq 1 ]] || "$@"; }
die()  { echo "ERREUR: $*" >&2; exit 1; }
warn() { echo "ATTENTION: $*" >&2; }

echo "=== Post-install $HOST — $(date) ==="
[[ $DRY   -eq 1 ]] && echo "*** MODE DRY-RUN : aucune modification ***"
[[ $FORCE -eq 1 ]] && echo "*** MODE FORCE ***"

# --- Identification machine (tracé dans le log) ------------------------------
echo "--- Identification ---"
echo "hostname (court) : $HOST"
echo "hostname (fqdn)  : $(hostname -f 2>/dev/null || echo 'n/a')"
echo "noyau            : $(uname -r)"
[[ -f /etc/os-release ]] && echo "OS               : $(. /etc/os-release; echo "$PRETTY_NAME")"
echo "--- Configuration reseau (ip a) ---"
ip a
echo "--- Table de routage (ip r) ---"
ip r || true
echo "-----------------------------------"

# --- 0. Garde-fou anti double passage ---------------------------------------
if [[ -f "$STAMP" && $FORCE -eq 0 ]]; then
    echo "Post-installation déjà effectuée sur cette machine :"
    cat "$STAMP"
    echo "Relancer avec --force pour rejouer."
    exit 0
fi

# --- 1. Contrôles préalables -------------------------------------------------
[[ $EUID -eq 0 ]]    || die "à lancer en root"
mountpoint -q "$KEY" || die "$KEY non monté"
[[ -d "$SRC_APP" ]]  || die "$SRC_APP introuvable"
[[ -f "$SRC_STARTX" ]] || die "$SRC_STARTX introuvable"
[[ -f "$SRC_SHADOW" ]] || die "$SRC_SHADOW introuvable"
id "$APP_USER" >/dev/null 2>&1 || die "compte $APP_USER inexistant"
getent group "$APP_GROUP" >/dev/null || die "groupe $APP_GROUP inexistant"
command -v md5sum >/dev/null || die "md5sum indisponible"

# --- 2. Autorisation d'écriture ---------------------------------------------
# 'mw' est un alias/fonction : on le rend disponible en mode non interactif.
if ! type mw >/dev/null 2>&1; then
    for f in /etc/profile.d/*.sh /root/.bashrc; do
        # shellcheck disable=SC1090
        [[ -f "$f" ]] && source "$f" || true
    done
fi
type mw >/dev/null 2>&1 || die "commande 'mw' indisponible — vérifier 'type mw' sur la machine"
run mw

# --- 3. Couche applicative ---------------------------------------------------
run mkdir -p "$DST_DATA"
run cp -a "$SRC_APP/." "$DST_DATA/"
run chown -R "$APP_USER:$APP_GROUP" "$DST_DATA"
run chmod "$DATA_MODE" "$DST_DATA"

# --- 4. Correctif ------------------------------------------------------------
run install -o root -g root -m 755 "$SRC_STARTX" "$DST_BIN/"

# --- 5. Mots de passe --------------------------------------------------------
run cp -a /etc/shadow "/etc/shadow.bak_${TS}"
if [[ $DRY -eq 0 ]]; then
    cat "$SRC_SHADOW" > /etc/shadow
else
    echo "+ cat $SRC_SHADOW > /etc/shadow"
fi
run chown root:root /etc/shadow
run chmod "$SHADOW_MODE" /etc/shadow
command -v restorecon >/dev/null && run restorecon -F /etc/shadow || true

# --- 6. Vérification par checksum -------------------------------------------
echo "--- Vérification md5 clé -> $DST_DATA ---"
RC_MD5=0
if [[ $DRY -eq 1 ]]; then
    echo "(ignorée en dry-run)"
else
    MD5LIST="$(mktemp)"
    ERRLIST="$(mktemp)"
    trap 'rm -f "$MD5LIST" "$ERRLIST"' EXIT

    # empreintes des fichiers source, en chemins relatifs
    ( cd "$SRC_APP" && find . -type f -print0 | sort -z | xargs -0 -r md5sum ) > "$MD5LIST"
    NB_SRC=$(wc -l < "$MD5LIST")

    # contrôle depuis la destination
    if ( cd "$DST_DATA" && md5sum -c --quiet "$MD5LIST" ) 2> "$ERRLIST" > "$ERRLIST.out"; then
        echo "OK : $NB_SRC/$NB_SRC fichiers conformes."
    else
        RC_MD5=1
        NB_KO=$(grep -c . "$ERRLIST.out" 2>/dev/null || echo 0)
        warn "$NB_KO fichier(s) en écart sur $NB_SRC :"
        head -n 20 "$ERRLIST.out" "$ERRLIST" || true
    fi
    rm -f "$ERRLIST.out"

    # fichiers présents en trop côté destination (hors fichier de contrôle)
    NB_DST=$(find "$DST_DATA" -type f | wc -l)
    [[ "$NB_DST" -ne "$NB_SRC" ]] && warn "nombre de fichiers différent : source=$NB_SRC destination=$NB_DST"
fi

# --- 7. Contrôles complémentaires -------------------------------------------
echo "--- Contrôles ---"
ls -l "$DST_BIN/$(basename "$SRC_STARTX")"
ls -l /etc/shadow
if [[ $DRY -eq 0 ]]; then
    pwck -r || warn "pwck signale des incohérences passwd/shadow — à vérifier"
fi

# --- 8. Horodatage de fin ----------------------------------------------------
if [[ $DRY -eq 0 && $RC_MD5 -eq 0 ]]; then
    mkdir -p "$(dirname "$STAMP")"
    {
        echo "machine        : $HOST"
        echo "date           : $(date '+%Y-%m-%d %H:%M:%S')"
        echo "operateur      : ${SUDO_USER:-root}"
        echo "adresses IP    : $(ip -o -4 addr show scope global | awk '{print $2"="$4}' | tr '\n' ' ')"
        echo "source NAVREP  : $SRC_APP"
        echo "destination    : $DST_DATA"
        echo "fichiers copies: ${NB_SRC:-?}"
        echo "md5            : OK"
        echo "log            : $LOG"
    } > "$STAMP"
    chmod 644 "$STAMP"
    echo "Jeton de post-installation écrit dans $STAMP"
elif [[ $RC_MD5 -ne 0 ]]; then
    warn "jeton NON écrit (écarts md5) — corriger puis relancer avec --force"
fi

echo "=== Terminé (md5=$([[ $RC_MD5 -eq 0 ]] && echo OK || echo KO)) — log : $LOG ==="
exit $RC_MD5
